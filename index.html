Step by step setup
1

Get Anthropic API key

Go to console.anthropic.com → Sign in → API Keys → Create Key → paste above.

2

Create Google Sheet

Go to sheets.google.com → Create new sheet → name it "Student Prompts".

3

Open Apps Script

In Google Sheets → click Extensions → Apps Script → delete existing code → paste this:

function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); var hw = data.hw || 'General'; // Get or create sheet tab for this homework var tab = sheet.getSheetByName(hw); if (!tab) { tab = sheet.insertSheet(hw); // Add headers var headers = ['Time', 'Name', 'Student ID']; for (var i = 1; i <= 20; i++) headers.push('Prompt ' + i); tab.appendRow(headers); tab.getRange(1, 1, 1, headers.length) .setFontWeight('bold') .setBackground('#e8f0fe'); } // Build row: time, name, id, then each prompt in its own column var row = [data.time, data.name, data.id]; var prompts = data.prompts || []; for (var j = 0; j < prompts.length; j++) { row.push(prompts[j]); } tab.appendRow(row); // Also log to master sheet var master = sheet.getSheetByName('All Submissions'); if (!master) { master = sheet.insertSheet('All Submissions'); master.appendRow(['Time','Name','ID','HW','Total Prompts','Prompt 1','Prompt 2','Prompt 3']); master.getRange(1,1,1,8).setFontWeight('bold').setBackground('#fce8e6'); } var masterRow = [data.time, data.name, data.id, hw, prompts.length]; for (var k = 0; k < Math.min(prompts.length, 10); k++) masterRow.push(prompts[k]); master.appendRow(masterRow); return ContentService .createTextOutput(JSON.stringify({status:'ok'})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({status:'error', message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
4

Deploy Apps Script

Click Deploy → New deployment → Web app
Execute as: Me | Access: Anyone
Click Deploy → copy the URL → paste in Setup above.

5

Host this file on GitHub Pages

1. Go to github.com → New repository
2. Upload this HTML file as index.html
3. Settings → Pages → Deploy from main branch
4. Your link: yourname.github.io/repo-name
